Complete Example

01_raw_sdk_loan_evaluator.py
import json
import os
from anthropic import Anthropic

# Initialize client (reads ANTHROPIC_API_KEY from environment)
client = Anthropic()

# Define the loan evaluation prompt
SYSTEM_PROMPT = """You are a loan evaluation system.

CRITERIA SCORING:
- Credit Score 750+: 40 points (Excellent)
- Credit Score 650-749: 25 points (Good)  
- Credit Score 600-649: 10 points (Fair)
- Credit Score <600: 0 points (Poor)

- Debt-to-Income Ratio <0.36: 40 points
- Debt-to-Income Ratio 0.36-0.43: 25 points
- Debt-to-Income Ratio >0.43: 0 points

- Loan Amount <50% of annual income: 20 points
- Loan Amount 50-100% of annual income: 10 points
- Loan Amount >100% of annual income: 0 points

DECISION RULES:
- Score 80+: APPROVED
- Score 50-79: REVIEW
- Score <50: DENIED

OUTPUT FORMAT (MUST BE VALID JSON):
{
  "decision": "approved|denied|review",
  "score": 0-100,
  "reasoning": "explain the decision",
  "factors": {
    "credit_score": score,
    "income_ratio": score,
    "loan_amount": score
  }
}"""

def evaluate_loan(applicant: dict) -> dict:
    """
    Evaluate a loan application using Claude.
    
    Args:
        applicant: Dict with credit_score, annual_income, loan_amount
        
    Returns:
        Dict with decision, score, reasoning
    """
    
    user_message = f"""Evaluate this loan application:
    
Credit Score: {applicant['credit_score']}
Annual Income: ${applicant['annual_income']:,}
Requested Loan: ${applicant['loan_amount']:,}
Debt-to-Income Ratio: {applicant['loan_amount'] / applicant['annual_income']:.2f}"""
    
    try:
        # Call Claude API directly
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=500,
            system=SYSTEM_PROMPT,
            messages=[
                {"role": "user", "content": user_message}
            ]
        )
        
        # Extract text from response
        response_text = response.content[0].text
        
        # Parse JSON response
        result = json.loads(response_text)
        
        return result
        
    except json.JSONDecodeError as e:
        return {
            "error": "Invalid JSON response",
            "raw_response": response_text
        }
    except Exception as e:
        return {"error": str(e)}

# Test cases: Good, Borderline, Risky
test_applications = [
    {
        "name": "Strong Candidate",
        "credit_score": 780,
        "annual_income": 150000,
        "loan_amount": 50000
    },
    {
        "name": "Borderline Case",
        "credit_score": 680,
        "annual_income": 65000,
        "loan_amount": 40000
    },
    {
        "name": "Risky Applicant",
        "credit_score": 580,
        "annual_income": 30000,
        "loan_amount": 50000
    }
]

# Run evaluations
if __name__ == "__main__":
    for app in test_applications:
        print(f"\n{'='*50}")
        print(f"Evaluating: {app['name']}")
        print(f"{'-'*50}")
        
        result = evaluate_loan(app)
        
        print(json.dumps(result, indent=2))

Key Concepts

1. No Abstraction

Raw SDK is direct API calls. You control every detail. This is the most flexible but also requires more code.

2. Error Handling

We catch two types of errors: JSON parsing errors (Claude's response isn't valid JSON) and API errors (network, timeout, etc.)

3. System vs User Messages

System: Permanent instructions (the rules)
User: The actual task (evaluate this application)

4. Response Structure

Claude's response is an object with content array. First element's text field contains the actual message.
