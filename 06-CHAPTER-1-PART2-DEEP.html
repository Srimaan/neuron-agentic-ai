4.2 Tutorial 1: Simple Agent (No Memory)

📚 Building a Research Agent from Scratch

Goal: Build an agent that searches the web for information and summarizes findings. Complexity: Beginner | Time: 20 minutes

Step 1: Define Tools

Python: Tool Definitions
import anthropic import json # Define the tools the agent can use TOOLS = [ { "name": "web_search", "description": "Search the web for information. Returns top 5 results with titles, URLs, and snippets.", "input_schema": { "type": "object", "properties": { "query": { "type": "string", "description": "The search query" } }, "required": ["query"] } }, { "name": "fetch_page", "description": "Fetch and read the full content of a webpage", "input_schema": { "type": "object", "properties": { "url": { "type": "string", "description": "The URL to fetch" } }, "required": ["url"] } } ] # Simulate tool execution def execute_tool(tool_name: str, tool_input: dict) -> str: """Simulate tool execution (in reality, call actual APIs)""" if tool_name == "web_search": # Simulated search results return json.dumps([ { "title": "Recent AI Breakthroughs 2024", "url": "https://example.com/ai-breakthrough", "snippet": "Anthropic releases Claude 4 with 200k context..." }, { "title": "OpenAI GPT-5 Released", "url": "https://example.com/gpt5", "snippet": "GPT-5 shows major improvements in reasoning..." } ]) elif tool_name == "fetch_page": return "Full page content for " + tool_input.get("url", "") else: return f"Unknown tool: {tool_name}"

Step 2: Implement the Agent Loop

Python: Complete Agent Loop
def run_research_agent(query: str, max_iterations: int = 10): """Run the agent loop""" client = anthropic.Anthropic() messages = [] iteration = 0 # Phase ①: Gather Context print(f"\n🔍 Starting research on: {query}") print("=" * 60) while iteration < max_iterations: iteration += 1 print(f"\n📍 Iteration {iteration}") # Add user message on first iteration if iteration == 1: messages.append({ "role": "user", "content": f"Research and summarize findings about: {query}" }) # Phase ②: Reason & Decide print("Thinking...") response = client.messages.create( model="claude-3-5-sonnet-20241022", max_tokens=1024, tools=TOOLS, messages=messages ) print(f"Stop reason: {response.stop_reason}") # Check if agent wants to use tools if response.stop_reason == "end_turn": # Phase ④: Reflect (Agent says we're done) print("✅ Agent finished reasoning") final_response = next( (block.text for block in response.content if hasattr(block, "text")), "No response" ) print(f"\n📝 Final Answer:\n{final_response}") return final_response elif response.stop_reason == "tool_use": # Phase ③: Take Action print("🔧 Executing tools...") messages.append({"role": "assistant", "content": response.content}) tool_results = [] for block in response.content: if block.type == "tool_use": print(f" → Calling {block.name}({block.input})") result = execute_tool(block.name, block.input) print(f" Result: {result[:100]}...") tool_results.append({ "type": "tool_result", "tool_use_id": block.id, "content": result }) messages.append({"role": "user", "content": tool_results}) else: print(f"Unexpected stop_reason: {response.stop_reason}") break print(f"\n⚠️ Max iterations ({max_iterations}) reached") return "Research incomplete - max iterations exceeded" # Run the agent result = run_research_agent("recent AI breakthroughs")

4.3 Tutorial 2: Agent with Persistent Memory (Database)

💾 Adding Production-Grade Memory

Challenge: Agent process crashes. How do we recover state? Solution: Persist messages to database. Reload on restart.

Step 1: Database Schema

SQL: Create tables for persistence
CREATE TABLE agent_sessions ( session_id VARCHAR(36) PRIMARY KEY, agent_id VARCHAR(100), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE agent_messages ( id INT AUTO_INCREMENT PRIMARY KEY, session_id VARCHAR(36), message_index INT, role VARCHAR(20), -- "user", "assistant" content LONGTEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (session_id) REFERENCES agent_sessions(session_id) ); CREATE TABLE agent_state ( session_id VARCHAR(36) PRIMARY KEY, iteration_count INT, total_tokens_used INT, is_complete BOOLEAN DEFAULT FALSE, last_tool_called VARCHAR(100), stop_reason VARCHAR(50), FOREIGN KEY (session_id) REFERENCES agent_sessions(session_id) );

Step 2: Persistent Agent Class

Python: Persistent Agent Implementation
import uuid import sqlite3 from datetime import datetime class PersistentAgent: def __init__(self, agent_id: str, db_path: str = "agent.db"): self.agent_id = agent_id self.db_path = db_path self.session_id = str(uuid.uuid4()) self.iteration = 0 self.messages = [] self.state = { "iteration_count": 0, "total_tokens_used": 0, "is_complete": False, "last_tool_called": None, "stop_reason": None } self._init_db() self._create_session() def _init_db(self): """Initialize database if needed""" conn = sqlite3.connect(self.db_path) cursor = conn.cursor() cursor.execute(""" CREATE TABLE IF NOT EXISTS agent_messages ( id INTEGER PRIMARY KEY, session_id TEXT, role TEXT, content TEXT, created_at TIMESTAMP ) """) cursor.execute(""" CREATE TABLE IF NOT EXISTS agent_state ( session_id TEXT PRIMARY KEY, iteration_count INTEGER, tokens_used INTEGER, is_complete BOOLEAN ) """) conn.commit() conn.close() def _create_session(self): """Create new session in database""" conn = sqlite3.connect(self.db_path) cursor = conn.cursor() cursor.execute(""" INSERT INTO agent_state (session_id, iteration_count, tokens_used, is_complete) VALUES (?, ?, ?, ?) """, (self.session_id, 0, 0, False)) conn.commit() conn.close() def add_message(self, role: str, content: str): """Add message AND persist to database""" msg = {"role": role, "content": content} self.messages.append(msg) # Persist immediately conn = sqlite3.connect(self.db_path) cursor = conn.cursor() cursor.execute(""" INSERT INTO agent_messages (session_id, role, content, created_at) VALUES (?, ?, ?, ?) """, (self.session_id, role, str(content), datetime.now())) conn.commit() conn.close() def update_state(self, **kwargs): """Update agent state""" self.state.update(kwargs) conn = sqlite3.connect(self.db_path) cursor = conn.cursor() cursor.execute(""" UPDATE agent_state SET iteration_count = ?, tokens_used = ?, is_complete = ? WHERE session_id = ? """, ( self.state["iteration_count"], self.state["total_tokens_used"], self.state["is_complete"], self.session_id )) conn.commit() conn.close() @staticmethod def load_session(session_id: str, db_path: str = "agent.db"): """Load previous session from database""" conn = sqlite3.connect(db_path) cursor = conn.cursor() # Load messages cursor.execute( "SELECT role, content FROM agent_messages WHERE session_id = ? ORDER BY id", (session_id,) ) messages = [{"role": r, "content": c} for r, c in cursor.fetchall()] # Load state cursor.execute( "SELECT iteration_count, tokens_used, is_complete FROM agent_state WHERE session_id = ?", (session_id,) ) state_row = cursor.fetchone() conn.close() agent = PersistentAgent.__new__(PersistentAgent) agent.session_id = session_id agent.messages = messages agent.state = { "iteration_count": state_row[0], "total_tokens_used": state_row[1], "is_complete": state_row[2] } if state_row else {} return agent # Usage example agent = PersistentAgent("research-agent-1") agent.add_message("user", "Research AI breakthroughs") agent.update_state(iteration_count=1, total_tokens_used=500) # If process crashes and restarts... # recovered_agent = PersistentAgent.load_session(session_id)

💡 Critical Pattern: Immediate Persistence

Notice we save to database IMMEDIATELY after each message. This means even if the process crashes mid-iteration, we can recover state. This is essential for production agents handling long-running tasks.

4.4 Common Pitfalls & Solutions

❌ Pitfall: Infinite Loops

Agent keeps calling same tool, getting same result, repeating.

Fix: Track tool calls, detect repetition:

if tool_name in last_3_tools: print("Loop detected, breaking") break

✅ Solution: Loop Detection

Keep history of recent tool calls. If same tool called 3x in a row, break loop.

tool_history = deque(maxlen=5) tool_history.append(tool_name) if len(set(tool_history)) == 1 and len(tool_history) >= 3: print("Infinite loop detected!") break

❌ Pitfall: Token Overflow

Context window grows too large (8 iterations × 2K tokens each = 16K).

✅ Solution: Summarization

After 5 iterations, summarize old messages and replace them.

Section 5: LangChain Agents (Quick Prototyping)

5.1 Tutorial 3: ReAct Agent with Memory

⚡ Build Agent in 20 Lines of Code

Python: Complete LangChain Agent
from langchain.agents import Tool, AgentExecutor, create_react_agent from langchain_anthropic import ChatAnthropic from langchain import hub from langchain_core.prompts import PromptTemplate from langchain.memory import ConversationBufferMemory # 1. Define tools def search_tool(query: str) -> str: return f"Search results for {query}: AI breakthroughs in 2024..." def calculate_tool(expression: str) -> str: try: return str(eval(expression)) except: return "Invalid expression" tools = [ Tool( name="Search", func=search_tool, description="Search for information about a topic" ), Tool( name="Calculator", func=calculate_tool, description="Calculate mathematical expressions" ) ] # 2. Create LLM llm = ChatAnthropic( model="claude-3-5-sonnet-20241022", temperature=0 ) # 3. Use built-in ReAct prompt prompt = hub.pull("hwchase17/react-chat") # 4. Create agent agent = create_react_agent(llm, tools, prompt) # 5. Add memory memory = ConversationBufferMemory( memory_key="chat_history", return_messages=True ) # 6. Create executor executor = AgentExecutor.from_agent_and_tools( agent, tools, memory=memory, verbose=True, max_iterations=10 ) # 7. Run! result = executor.invoke({"input": "What are recent AI breakthroughs? Calculate 2+2"}) print(result["output"])

💡 LangChain Advantage: Less Code

Compare to Claude SDK: LangChain handles the loop for you. Memory management is built-in. Trade-off: Less transparency on what's happening inside.

5.2 Understanding LangChain Memory Types

Python: Three Memory Strategies
from langchain.memory import ( ConversationBufferMemory, # Keep all messages ConversationSummaryMemory, # Summarize old messages ConversationEntityMemory # Extract and track entities ) # Strategy 1: Buffer (Simple) buffer_memory = ConversationBufferMemory( memory_key="history", return_messages=True ) # Problem: Context window grows indefinitely # Strategy 2: Summary (Smart) summary_memory = ConversationSummaryMemory( llm=llm, # Use LLM to summarize memory_key="history" ) # Each 5 messages get summarized: "User asked X, agent answered Y" # Strategy 3: Entity (Very Smart) entity_memory = ConversationEntityMemory( llm=llm, memory_key="history" ) # Extracts facts about entities: "User: Alice, Age: 25, Job: Engineer" # Retrieves only relevant entities to context

⚠️ Memory Gotcha: LangChain Doesn't Persist by Default

LangChain memory objects live only in RAM. Process restart = memory lost. For production, wrap memory with database: save after each iteration.

Section 6: LangGraph (State Machines for Complex Workflows)

6.1 Tutorial 4: Multi-Step Agent with Conditional Logic

🔀 Building Graph-Based Agents

Python: LangGraph State Machine
from langgraph.graph import StateGraph, START, END from typing import TypedDict, Annotated import operator # Define state structure class AgentState(TypedDict): messages: Annotated[list, operator.add] current_step: str research_complete: bool summary: str # Define nodes (functions) def gather_info_node(state: AgentState): """Phase 1: Gather context""" print("📍 Gathering information...") return { "messages": [{"role": "system", "content": "Gathering data..."}], "current_step": "research" } def reasoning_node(state: AgentState): """Phase 2: Reason about what to do""" print("📍 Reasoning...") return { "messages": [{"role": "assistant", "content": "I will search for X"}], "current_step": "action" } def action_node(state: AgentState): """Phase 3: Take action""" print("📍 Executing tool...") return { "messages": [{"role": "tool", "content": "Tool result..."}], "current_step": "reflection" } def reflection_node(state: AgentState): """Phase 4: Reflect and decide""" print("📍 Reflecting...") # Decide: continue or done? done = True # For demo return { "research_complete": done, "current_step": "summarize" if done else "reasoning" } def summarize_node(state: AgentState): """Final step: Summarize findings""" print("📍 Summarizing...") return { "summary": "Summary of findings...", "current_step": "end" } # Build graph graph = StateGraph(AgentState) # Add nodes graph.add_node("gather", gather_info_node) graph.add_node("reasoning", reasoning_node) graph.add_node("action", action_node) graph.add_node("reflection", reflection_node) graph.add_node("summarize", summarize_node) # Add edges graph.add_edge(START, "gather") graph.add_edge("gather", "reasoning") graph.add_edge("reasoning", "action") graph.add_edge("action", "reflection") # Conditional edge: Should we loop or finish? def should_continue(state: AgentState): if state["research_complete"]: return "summarize" else: return "reasoning" graph.add_conditional_edges("reflection", should_continue) graph.add_edge("summarize", END) # Compile compiled_graph = graph.compile() # Run result = compiled_graph.invoke({ "messages": [], "current_step": "start", "research_complete": False }) print(f"\n✅ Final summary: {result['summary']}")

💡 LangGraph Advantage: Visualization & Control

You can draw the graph and see it visually. Every decision is explicit. Debugging is easier because you know exactly which node you're in and why.
